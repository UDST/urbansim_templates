# dependencies (the floors declared in pyproject.toml). Used by the
# minimum-dependencies CI job:

#   pip install ".[test]" -c requirements-min.txt

# numpy must be constrained alongside pandas 1.5: pandas 1.5 predates
# NumPy 2 and declares no upper bound, so without this constraint the
# resolver pairs it with NumPy 2.x and pandas fails to import. tables is
# an Orca dependency and is pinned so the lower-bound environment stays
# reproducible.
